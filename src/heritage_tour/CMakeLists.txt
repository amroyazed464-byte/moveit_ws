cmake_minimum_required(VERSION 3.8)
project(heritage_tour)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
add_executable(scene_transport src/scene_transport.cpp)
target_link_libraries(scene_transport gz-transport13::core gz-msgs10::core)
install(TARGETS scene_transport DESTINATION lib/${PROJECT_NAME})
add_library(heritage_tour_screens SHARED src/screen_system.cpp)
target_link_libraries(heritage_tour_screens gz-sim8::core gz-transport13::core gz-msgs10::core)
install(TARGETS heritage_tour_screens LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch maps web voices DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/tour_runtime.py scripts/start_tour.sh DESTINATION lib/${PROJECT_NAME})
ament_package()
